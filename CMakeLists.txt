# Shared native telemetry core with separate Win32 and GTK desktop frontends.
cmake_minimum_required(VERSION 3.22)
project(NodeSight VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)

# Embed the same reviewed POSIX script for both platforms; no runtime file is installed remotely.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/collectors/linux.sh" COLLECTOR_SCRIPT)
configure_file(src/collector_script.h.in generated/collector_script.h @ONLY)
add_library(nodesight-core STATIC src/config.cpp src/metrics.cpp src/recording.cpp src/monitor.cpp src/dashboard.cpp)
target_include_directories(nodesight-core PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(nodesight-core PUBLIC Threads::Threads)

if(WIN32)
  enable_language(RC)
  target_sources(nodesight-core PRIVATE src/remote.cpp src/local_win.cpp)
  target_compile_definitions(nodesight-core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(nodesight-core PUBLIC shell32 advapi32 iphlpapi psapi ole32)
  add_executable(NodeSight WIN32 src/main.cpp src/settings_win.cpp src/app.rc)
  target_link_libraries(NodeSight PRIVATE nodesight-core user32 gdi32 comctl32 comdlg32 dwmapi uxtheme)
  add_executable(node-sight-askpass WIN32 src/askpass.cpp)
  target_compile_definitions(node-sight-askpass PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(node-sight-askpass PRIVATE advapi32)
  if(MSVC)
    # The resource already contains the DPI/privilege manifest.
    target_link_options(NodeSight PRIVATE /MANIFEST:NO)
  endif()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
  target_sources(nodesight-core PRIVATE src/remote_linux.cpp src/local_linux.cpp)
  add_executable(NodeSight src/main_linux.cpp)
  target_link_libraries(NodeSight PRIVATE nodesight-core PkgConfig::GTK3)
else()
  message(FATAL_ERROR "Node-Sight supports Windows and Linux desktop clients.")
endif()

add_executable(metrics-tests tests/metrics_test.cpp)
target_link_libraries(metrics-tests PRIVATE nodesight-core)
add_executable(core-tests tests/core_test.cpp)
target_link_libraries(core-tests PRIVATE nodesight-core)
foreach(target nodesight-core NodeSight metrics-tests core-tests)
  if(MSVC)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive- /EHsc $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/GL>)
    target_link_options(${target} PRIVATE $<$<CONFIG:Release>:/LTCG> $<$<CONFIG:Release>:/OPT:REF>)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(MSVC)
  set_property(TARGET node-sight-askpass PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(node-sight-askpass PRIVATE /W4 /utf-8 /O1)
endif()
enable_testing()
add_test(NAME metrics COMMAND metrics-tests)
add_test(NAME configuration-and-recording COMMAND core-tests)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # The fixture is confined to this test's PATH; it never alters a system SSH executable.
  configure_file(tests/fake-ssh/ssh test-ssh/ssh COPYONLY)
  file(CHMOD "${CMAKE_CURRENT_BINARY_DIR}/test-ssh/ssh" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
  add_executable(transport-tests tests/transport_test.cpp)
  target_link_libraries(transport-tests PRIVATE nodesight-core)
  add_test(NAME concurrent-transports COMMAND transport-tests "${CMAKE_CURRENT_BINARY_DIR}/test-ssh")
endif()
if(WIN32)
  install(TARGETS NodeSight RUNTIME DESTINATION .)
  install(TARGETS node-sight-askpass RUNTIME DESTINATION .)
else()
  install(TARGETS NodeSight RUNTIME DESTINATION bin)
  install(FILES packaging/node-sight.desktop DESTINATION share/applications)
endif()
